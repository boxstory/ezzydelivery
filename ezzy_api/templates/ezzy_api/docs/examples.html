💻 Code Examples

Ready-to-use integration examples

Python Example

import requests
import os

api_key = os.getenv('EZZYDELIVERY_API_KEY')
base_url = "https://api.ezzydelivery.qa"

headers = {
  "Authorization": f"Bearer {api_key}",
  "Content-Type": "application/json"
}

# Import Shopify orders
payload = {
  "store_url": "mystore.myshopify.com",
  "order_ids": [123456, 123457]
}

response = requests.post(
  f"{base_url}/api/integrations/shopify/import/",
  json=payload,
  headers=headers
)

if response.status_code == 200:
  print("Orders imported successfully")
else:
  print(f"Error: {response.text}")

JavaScript/Node.js Example

const axios = require('axios');

const apiKey = process.env.EZZYDELIVERY_API_KEY;
const baseUrl = 'https://api.ezzydelivery.qa';

const headers = {
  Authorization: `Bearer ${apiKey}`,
  'Content-Type': 'application/json'
};

async function importOrders() {
  try {
    const response = await axios.post(
      `${baseUrl}/api/integrations/shopify/import/`,
      {
        store_url: 'mystore.myshopify.com',
        order_ids: [123456, 123457]
      },
      { headers }
    );
    console.log('Success:', response.data);
  } catch (error) {
    console.error('Error:', error.response.data);
  }
}

importOrders();
